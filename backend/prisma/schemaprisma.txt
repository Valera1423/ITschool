model Profile {
  id          String    @id @db.Uuid
  email       String    @unique
  fullName    String    @map("full_name")
  role        String    @default("nam") @db.VarChar(20) // admin | manager | nam
  avatarColor String    @default("blue") @map("avatar_color")
  blocked     Boolean   @default(false)
  totpEnabled Boolean   @default(false) @map("totp_enabled")
  keycloakSub String?   @unique @map("keycloak_sub")
  lastLoginAt DateTime? @map("last_login_at")
  createdAt   DateTime  @default(now()) @map("created_at")
  updatedAt   DateTime  @updatedAt @map("updated_at")
  customRoleId String?    @map("custom_role_id") @db.Uuid
  customRole   CustomRole? @relation(fields: [customRoleId], references: [id], onDelete: SetNull)

  googleOAuth       GoogleOAuthToken?
  contracts         Contract[]
  auditLogs         AuditLog[]
  notifications     Notification[]
  pushSubscriptions PushSubscription[]
  sessions          UserSession[]
  preferences       UserPreferences?
  scopes            UserScope[]

  @@map("profiles")
}

model UserScope {
  id        String   @id @default(uuid()) @db.Uuid
  userId    String   @map("user_id") @db.Uuid
  entity    String   @default("universities")
  filter    Json     @default("{}")
  createdAt DateTime @default(now()) @map("created_at")
  updatedAt DateTime @updatedAt @map("updated_at")

  user Profile @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([userId, entity])
  @@index([entity])
  @@map("user_scopes")
}

// ============================================================
// Правила переходов между шагами workflow
// ============================================================
model WorkflowTransitionRule {
  id         String   @id @default(uuid()) @db.Uuid
  workflowId String   @map("workflow_id") @db.Uuid
  fromStepId String   @map("from_step_id") @db.Uuid
  toStepId   String   @map("to_step_id") @db.Uuid
  /**
   * Условие необязательно. Формат — JSON-объект вида
   *   { field: "status", operator: "not_equals", value: "Отказ" }
   * Проверяется в POST /workflows/transitions, если задано.
   */
  condition  Json     @default("{}")
  /** Пометка: требуется комментарий (по ТЗ п.2 он обязателен, но
   *  иногда хочется пометить «важный» переход отдельно). */
  requiresComment Boolean @default(true) @map("requires_comment")
  createdAt  DateTime @default(now()) @map("created_at")

  workflow Workflow     @relation(fields: [workflowId], references: [id], onDelete: Cascade)
  fromStep WorkflowStep @relation("RuleFromStep", fields: [fromStepId], references: [id], onDelete: Cascade)
  toStep   WorkflowStep @relation("RuleToStep",   fields: [toStepId],   references: [id], onDelete: Cascade)

  @@unique([workflowId, fromStepId, toStepId])
  @@index([workflowId])
  @@index([fromStepId])
  @@index([toStepId])
  @@map("workflow_transition_rules")
}
